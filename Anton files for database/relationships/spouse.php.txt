<?php

session_start();

require_once "../config/database.php";

$message = "";

if ($_SERVER["REQUEST_METHOD"] === "POST") {

    $parent_id = intval($_POST["parent_id"]);
    $child_id = intval($_POST["child_id"]);

    $relationship_type =
        $_POST["relationship_type"] ?? "Biological";

    $parent_role =
        $_POST["parent_role"] ?? "Unknown";

    $confidence =
        $_POST["confidence"] ?? "Confirmed";

    $notes =
        trim($_POST["notes"] ?? "");

    if ($parent_id === $child_id) {

        $message = "A person cannot be their own parent.";

    } else {

        try {

            $stmt = $pdo->prepare("
                INSERT INTO parent_child_relationships
                (
                    parent_id,
                    child_id,
                    relationship_type,
                    parent_role,
                    confidence,
                    notes
                )
                VALUES (?, ?, ?, ?, ?, ?)
            ");

            $stmt->execute([
                $parent_id,
                $child_id,
                $relationship_type,
                $parent_role,
                $confidence,
                $notes ?: null
            ]);

            $message = "Parent-child relationship saved.";

        } catch (PDOException $e) {

            $message = "Relationship could not be saved.";

        }
    }
}

$people = $pdo->query("
    SELECT
        id,
        first_name,
        middle_name,
        last_name
    FROM people
    ORDER BY last_name, first_name
")->fetchAll();

<?php endif; ?>

</option>

        $person["first_name"] . " " .

        ($person["last_name"] ?? "")
    )
) ?>

</option>

<?php endforeach; ?>

</select>


<label>Child</label>

<select name="child_id" required>

<option value="">
Select child
</option>

<?php foreach ($people as $person): ?>

    trim(
        $person["first_name"] . " " .
        ($person["middle_name"] ?? "") . " " .
        ($person["last_name"] ?? "")
    )
) ?>
