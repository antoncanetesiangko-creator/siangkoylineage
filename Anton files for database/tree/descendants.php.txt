<?php

require_once "../config/database.php";

$id = intval($_GET["id"] ?? 0);

if (!$id) {
    die("Person ID required.");
}

$stmt = $pdo->prepare("
    WITH RECURSIVE descendants AS (

        SELECT
            p.id,
            p.first_name,
            p.middle_name,
            p.last_name,
            p.sex,
            0 AS generation

        FROM people p

        WHERE p.id = ?

        UNION ALL

        SELECT
            child.id,
            child.first_name,
            child.middle_name,
            child.last_name,
            child.sex,
            descendants.generation + 1

        FROM descendants

        INNER JOIN parent_child_relationships pc
            ON pc.parent_id = descendants.id

        INNER JOIN people child
            ON child.id = pc.child_id

    )

    SELECT *
    FROM descendants

    ORDER BY generation, last_name, first_name
");

$stmt->execute([$id]);

$descendants = $stmt->fetchAll();

?>

<!DOCTYPE html>

<html>

<head>

<title>Descendants</title>

<link rel="stylesheet"
      href="../assets/css/style.css">

</head>

<body>

<div class="container">

<h1>Descendants</h1>

<?php foreach ($descendants as $person): ?>

<?php

$indent = $person["generation"] * 40;

$name = trim(
    $person["first_name"] . " " .
    ($person["middle_name"] ?? "") . " " .
    ($person["last_name"] ?? "")
);

?>

<div
    class="tree-person"
    style="margin-left: <?= $indent ?>px;"
>

<a href="../people/view.php?id=<?= $person["id"] ?>">

<?= htmlspecialchars($name) ?>

</a>

<span>
Generation <?= $person["generation"] ?>
</span>

</div>

<?php endforeach; ?>

</div>

</body>

</html>
