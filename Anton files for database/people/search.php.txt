<?php

session_start();

require_once "../config/database.php";

$id = intval($_GET["id"] ?? 0);

if ($id <= 0) {
    die("Invalid person ID.");
}

$stmt = $pdo->prepare("
    SELECT
        p.*,
        bp.place_name AS birth_place,
        dp.place_name AS death_place
    FROM people p

    LEFT JOIN places bp
        ON p.birth_place_id = bp.id

    LEFT JOIN places dp
        ON p.death_place_id = dp.id

    WHERE p.id = ?
");

$stmt->execute([$id]);

$person = $stmt->fetch();

if (!$person) {
    die("Person not found.");
}

$parentsStmt = $pdo->prepare("
    SELECT
        p.id,
        p.first_name,
        p.middle_name,
        p.last_name,
        p.sex,
        r.relationship_type,
        r.parent_role

    FROM parent_child_relationships r

    INNER JOIN people p
        ON p.id = r.parent_id

    WHERE r.child_id = ?

    ORDER BY
        FIELD(r.parent_role, 'Father', 'Mother', 'Parent', 'Unknown')
");

$parentsStmt->execute([$id]);

$parents = $parentsStmt->fetchAll();

$childrenStmt = $pdo->prepare("
    SELECT
        p.id,
        p.first_name,
        p.middle_name,
        p.last_name,
        p.sex,
        r.relationship_type,
        r.parent_role

    FROM parent_child_relationships r

    INNER JOIN people p
        ON p.id = r.child_id

    WHERE r.parent_id = ?

    ORDER BY p.last_name, p.first_name
");

$childrenStmt->execute([$id]);

$children = $childrenStmt->fetchAll();

$mediaStmt = $pdo->prepare("
    SELECT *
    FROM media
    WHERE person_id = ?
    ORDER BY uploaded_at DESC
");

$mediaStmt->execute([$id]);

$media = $mediaStmt->fetchAll();

$storiesStmt = $pdo->prepare("
    SELECT *
    FROM family_stories
    WHERE person_id = ?
    ORDER BY created_at DESC
");

$storiesStmt->execute([$id]);

$stories = $storiesStmt->fetchAll();
